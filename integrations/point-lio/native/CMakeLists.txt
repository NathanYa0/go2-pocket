cmake_minimum_required(VERSION 3.20)
project(pointlio_mac LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/../../../output/point-lio/ws/src/point_lio_unilidar" CACHE PATH "Pinned upstream source")
find_package(PCL REQUIRED COMPONENTS common io filters)
find_package(yaml-cpp CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_library(OMP_LIBRARY omp PATHS /opt/homebrew/opt/libomp/lib REQUIRED)
# Patch only a build-directory copy for modern Eigen/Apple Clang alignment.
file(COPY "${UPSTREAM}/src" "${UPSTREAM}/include" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/upstream")
set(NATIVE_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/upstream")
file(READ "${NATIVE_SOURCE}/src/preprocess.h" preprocess)
string(REPLACE "namespace unilidar_ros {\nstruct Point" "namespace unilidar_ros {\nstruct EIGEN_ALIGN16 Point" preprocess "${preprocess}")
string(REPLACE "} EIGEN_ALIGN16;" "};" preprocess "${preprocess}")
file(WRITE "${NATIVE_SOURCE}/src/preprocess.h" "// Modified for Apple Clang/Eigen alignment by Go2 Pocket, 2026-11-06. GPL-2.0.\n${preprocess}")
file(READ "${NATIVE_SOURCE}/src/laserMapping.cpp" mapping)
string(REPLACE "map_incremental();" "if (ikdtree.size() > 2000000) throw std::runtime_error(\"Native map limit reached; start a new session\");\n            map_incremental();" mapping "${mapping}")
file(WRITE "${NATIVE_SOURCE}/src/laserMapping.cpp" "// Modified with native map-memory guard by Go2 Pocket, 2026-11-06. GPL-2.0.\n${mapping}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/shim")
foreach(header ros/ros.h sensor_msgs/Imu.h sensor_msgs/PointCloud2.h nav_msgs/Odometry.h nav_msgs/Path.h geometry_msgs/Vector3.h tf/transform_datatypes.h tf/transform_broadcaster.h eigen_conversions/eigen_msg.h pcl_conversions/pcl_conversions.h visualization_msgs/Marker.h)
  get_filename_component(dir "${header}" DIRECTORY)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/shim/${dir}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shim/${header}" "#include \"native.hpp\"\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shim/Python.h" "// Upstream includes Python.h but uses no Python API.\n")
add_executable(pointlio-mac main.cpp "${NATIVE_SOURCE}/src/laserMapping.cpp" "${NATIVE_SOURCE}/src/Estimator.cpp" "${NATIVE_SOURCE}/src/parameters.cpp" "${NATIVE_SOURCE}/src/preprocess.cpp" "${NATIVE_SOURCE}/include/ikd-Tree/ikd_Tree.cpp")
set_source_files_properties("${NATIVE_SOURCE}/src/laserMapping.cpp" PROPERTIES COMPILE_DEFINITIONS "main=point_lio_main")
target_include_directories(pointlio-mac PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/shim" "${NATIVE_SOURCE}/include" "${NATIVE_SOURCE}/src" /opt/homebrew/opt/libomp/include ${PCL_INCLUDE_DIRS})
target_compile_definitions(pointlio-mac PRIVATE ROOT_DIR="${UPSTREAM}/")
target_compile_options(pointlio-mac PRIVATE -include native.hpp -O2 -Wno-deprecated-declarations -Wno-unknown-pragmas)
target_link_libraries(pointlio-mac PRIVATE ${PCL_LIBRARIES} yaml-cpp::yaml-cpp nlohmann_json::nlohmann_json "${OMP_LIBRARY}")
